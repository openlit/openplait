# Python

OpenPlait’s Python package lives in [`python/`](https://github.com/openlit/openplait/tree/main/python)
and mirrors the TypeScript workspace under [`typescript/`](https://github.com/openlit/openplait/tree/main/typescript).

| Python module | TypeScript package |
| --- | --- |
| `openplait.core` | `@openplait/core` |
| `openplait.adapter_sdk` | `@openplait/adapter-sdk` |
| `openplait.runtime` | `@openplait/runtime` |
| `openplait.adapters.clickhouse` | `@openplait/adapter-clickhouse` |
| `openplait.adapters.tempo` | `@openplait/adapter-tempo` |
| `openplait.adapters.loki` | `@openplait/adapter-loki` |
| `openplait.adapters.prometheus` | `@openplait/adapter-prometheus` |
| `openplait.adapters.jaeger` | `@openplait/adapter-jaeger` |

Status: **1.0.0**. Config validation, capabilities, runtime registration, and
**native execute** work for all five shipped connectors. TypeScript remains the
reference for full semantic compile parity (ClickHouse SQL compiler, Tempo
attribute mapping, etc.).

## Register connectors

Never put passwords or tokens in query documents — only in server-side config.

<LangTabs ariaLabel="Register connectors language">
  <LangTab label="Python">

```python
from openplait.adapters import (
    ClickHouseAdapter,
    ClickHouseConfig,
    JaegerAdapter,
    JaegerConfig,
    LokiAdapter,
    LokiConfig,
    PrometheusAdapter,
    PrometheusConfig,
    TempoAdapter,
    TempoConfig,
)
from openplait.runtime import DatasourceRegistry, OpenPlaitRuntime
from openplait.runtime.registry import RuntimeDatasourceRegistration

registry = DatasourceRegistry()
registry.register(
    RuntimeDatasourceRegistration(
        name="primary",
        kind="ClickHouseDatasource",
        adapter=ClickHouseAdapter(ClickHouseConfig(url="http://localhost:8123")),
        config=ClickHouseConfig(url="http://localhost:8123", database="openlit"),
    )
)
registry.register(
    RuntimeDatasourceRegistration(
        name="tempo",
        kind="TempoDatasource",
        adapter=TempoAdapter(TempoConfig(url="http://localhost:3200")),
        config=TempoConfig(url="http://localhost:3200"),
    )
)
registry.register(
    RuntimeDatasourceRegistration(
        name="logs",
        kind="LokiDatasource",
        adapter=LokiAdapter(LokiConfig(url="http://localhost:3100")),
        config=LokiConfig(url="http://localhost:3100"),
    )
)
registry.register(
    RuntimeDatasourceRegistration(
        name="metrics",
        kind="PrometheusDatasource",
        adapter=PrometheusAdapter(PrometheusConfig(url="http://localhost:9090")),
        config=PrometheusConfig(url="http://localhost:9090"),
    )
)
registry.register(
    RuntimeDatasourceRegistration(
        name="jaeger",
        kind="JaegerDatasource",
        adapter=JaegerAdapter(JaegerConfig(url="http://localhost:16686")),
        config=JaegerConfig(url="http://localhost:16686"),
    )
)
runtime = OpenPlaitRuntime(registry)
```

  </LangTab>
  <LangTab label="TypeScript">

```ts
import { ClickHouseAdapter } from "@openplait/adapter-clickhouse";
import { TempoAdapter } from "@openplait/adapter-tempo";
import { LokiAdapter } from "@openplait/adapter-loki";
import { PrometheusAdapter } from "@openplait/adapter-prometheus";
import { JaegerAdapter } from "@openplait/adapter-jaeger";
import { DatasourceRegistry, OpenPlaitRuntime } from "@openplait/runtime";

const registry = new DatasourceRegistry()
  .register({
    name: "primary",
    kind: "ClickHouseDatasource",
    scope: "workspace",
    config: { url: "http://localhost:8123" },
    adapter: new ClickHouseAdapter({ url: "http://localhost:8123" }),
  })
  .register({
    name: "tempo",
    kind: "TempoDatasource",
    scope: "workspace",
    config: { url: "http://localhost:3200" },
    adapter: new TempoAdapter({ url: "http://localhost:3200" }),
  });

const runtime = new OpenPlaitRuntime(registry);
```

</LangTabs>

## Native execute example (Loki)

<LangTabs ariaLabel="Native Loki language">
  <LangTab label="Python">

```python
from openplait import OPENPLAIT_API_VERSION
from openplait.core import parse_query

query = parse_query(
    {
        "apiVersion": OPENPLAIT_API_VERSION,
        "kind": "Query",
        "metadata": {"name": "application-errors"},
        "spec": {
            "mode": "native",
            "datasource": {"kind": "LokiDatasource", "name": "logs"},
            "native": {
                "language": "logql",
                "statement": '{service_name="checkout"} |= "failed"',
            },
            "extensions": {
                "io.openplait.loki": {
                    "timeRange": {
                        "from": "2026-08-05T00:00:00Z",
                        "to": "2026-08-05T01:00:00Z",
                    },
                    "limit": 500,
                }
            },
        },
    }
)
result = await runtime.execute(query)
```

  </LangTab>
  <LangTab label="TypeScript">

```ts
const result = await runtime.execute({
  apiVersion: "openplait.io/v1alpha1",
  kind: "Query",
  metadata: { name: "application-errors" },
  spec: {
    mode: "native",
    datasource: { kind: "LokiDatasource", name: "logs" },
    native: {
      language: "logql",
      statement: '{service_name="checkout"} |= "failed"',
    },
    extensions: {
      "io.openplait.loki": {
        timeRange: {
          from: "2026-08-05T00:00:00Z",
          to: "2026-08-05T01:00:00Z",
        },
        limit: 500,
      },
    },
  },
});
```
