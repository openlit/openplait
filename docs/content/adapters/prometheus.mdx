# Prometheus Adapter

Executes bounded `/api/v1/query_range` requests and normalizes matrix, vector,
and scalar responses into timestamp, value, and label fields. Works with
Prometheus-compatible APIs such as Grafana Mimir.

- **TypeScript:** `@openplait/adapter-prometheus` — semantic + native PromQL
- **Python:** `openplait.adapters.PrometheusAdapter` — native PromQL `query_range`

## Configure

<LangTabs ariaLabel="Configure Prometheus language">
  <LangTab label="TypeScript">

```ts
import { PrometheusAdapter } from "@openplait/adapter-prometheus";

const adapter = new PrometheusAdapter({
  url: "http://localhost:9090",
  allowNativeQueries: true,
  defaultStep: "30s",
  labelFields: { "service.name": "service_name" },
});
```

  </LangTab>
  <LangTab label="Python">

```python
from openplait.adapters import PrometheusAdapter, PrometheusConfig

adapter = PrometheusAdapter(
    PrometheusConfig(
        url="http://localhost:9090",
        allow_native_queries=True,
        default_step="30s",
    )
)
```

  </LangTab>
</LangTabs>

## Native PromQL

<LangTabs ariaLabel="Native Prometheus query language">
  <LangTab label="TypeScript">

```ts
const query = {
  apiVersion: "openplait.io/v1alpha1",
  kind: "Query",
  metadata: { name: "request-rate" },
  spec: {
    mode: "native",
    datasource: { kind: "PrometheusDatasource", name: "metrics" },
    native: {
      language: "promql",
      statement: 'sum(rate(http_requests_total[5m])) by (service_name)',
    },
    extensions: {
      "io.openplait.prometheus": {
        timeRange: { from: "2026-08-05T00:00:00Z", to: "2026-08-05T01:00:00Z" },
        step: "30s",
      },
    },
  },
};
```

  </LangTab>
  <LangTab label="Python">

```python
from openplait import OPENPLAIT_API_VERSION
from openplait.core import parse_query

query = parse_query(
    {
        "apiVersion": OPENPLAIT_API_VERSION,
        "kind": "Query",
        "metadata": {"name": "request-rate"},
        "spec": {
            "mode": "native",
            "datasource": {"kind": "PrometheusDatasource", "name": "metrics"},
            "native": {
                "language": "promql",
                "statement": 'sum(rate(http_requests_total[5m])) by (service_name)',
            },
            "extensions": {
                "io.openplait.prometheus": {
                    "timeRange": {
                        "from": "2026-08-05T00:00:00Z",
                        "to": "2026-08-05T01:00:00Z",
                    },
                    "step": "30s",
                }
            },
        },
    }
)
```

</LangTabs>
